\section*{Setup and Notation}

We consider a sample of $n$ i.i.d.\ subjects $(X_i, W_i, Y_i(1), Y_i(0)) \sim \mathbb{P}$, where $W_i \in \{0,1\}$ is a scalar treatment, $X_i \in \mathbb{R}^p$ a covariate vector, and $Y_i(w) \in \mathbb{R}$ potential outcomes.  We observe only $Y_i = Y_i(W_i)$ and write $Z_i = (X_i, W_i, Y_i)$.

We define $e(x) = \mathbb{P}(W_i=1\mid X_i=x)$, $m_w(x) = \mathbb{E}[Y_i\mid W_i=w, X_i=x]$, and $\tau(x) = \log\bigl(m_1(x)/m_0(x)\bigr)$.

The target estimand is
\[
  \tau = \log\frac{\mu_1}{\mu_0},
  \qquad
  \mu_w = \mathbb{E}\bigl[\phi(X, P)\,Y(w)\bigr].
\]
The weight $\phi(x, s, t)$ \emph{depends on the unknown distribution $P$} through the scalar $\tau$ and the
conditional effect function $\tau(\cdot)$, where $s$ is a placeholder for the scalar $\tau$ and $t$ is a placeholder for the pointwise value $\tau(x)$; the natural evaluation is $\phi(x,\tau,\tau(x))$.  Partial
derivatives are
\[
  \phi_\tau'(x) := \frac{\partial \phi(x,s,t)}{\partial s}\bigg|_{s=\tau,\,t=\tau(x)}, \qquad
  \phi_{\tau(\cdot)}'(x) := \frac{\partial \phi(x, s, t)}{\partial t}\bigg|_{s=\tau,\,t=\tau(x)}.
\]
That is, $\phi_\tau'(x)$ is the derivative of $\phi$ with respect to the scalar estimand $\tau$, while
$\phi_{\tau(\cdot)}'(x)$ is the derivative with respect to the \emph{pointwise} value $\tau(x)$---treating $\phi$ as a function of the third argument $t$ and evaluating at $t=\tau(x)$.
We treat $e(x)$ and the marginal density $p(x)$ as known, so the only unknown nuisance is the conditional
mean pair $(m_0, m_1)$.


\paragraph{Problem:} We are looking to find a semi-parametric efficiency bound for $\tau$ when the weight $\phi$ is allowed to depend on the unknown $P$ through $\tau$ and $\tau(\cdot)$. We are also looking to find the weights, for which this bound is minimal. Specifically, for any submodel $P_\epsilon$:

\[V(\hat \tau) \geq \left(\frac{d\tau}{d\epsilon}\bigg|_0\right)^2\frac{1}{\mathbb{E}[s(z_i)^2]}.\]

\paragraph{Minimax optimal weights:} We then ask which weight family minimises the worst-case efficiency loss relative to the oracle weight, over $\tau$, over $m_0$, and over local departures from $\tau(\cdot)\equiv\tau$; see Theorem \ref{thm:minimax}.

\begin{assumption}[Known $e(x)$ and $p(x)$]
The propensity score $e(\cdot)$ and the marginal density $p(\cdot)$ are known, so the only unknown nuisance is $(m_0,m_1)$.
\end{assumption}

\begin{assumption}[Unconfoundedness and overlap]
$W \perp\!\!\!\perp (Y(1),Y(0))\mid X$ and $e(x)\in(\eta,1-\eta)$ for some $\eta>0$.
\end{assumption}

Under unconfoundedness we can identify $\tau$ as a functional of the observed data distribution:

\[
  \tau = \log\mu_1 - \log\mu_0,
  \qquad
  \mu_w = \mathbb{E}\bigl[\phi(X, P)\,Y(w)\bigr].
\]

\begin{assumption}[Homoskedasticity]
\[\mathbb{V}[Y\mid X,W] = \sigma^2\,\mathbb{E}[Y\mid X,W] \]
\end{assumption}

Considering the following parametric submodel:

\[
  \epsilon \mapsto f_{Y\mid X,W}(y \mid x, 1; \epsilon) = f_{Y\mid X,W}(y \mid x, 1)\bigl(1 + \epsilon c_1(y,x)\bigr),
\]
and
\[
  \epsilon \mapsto f_{Y\mid X,W}(y \mid x, 0; \epsilon) = f_{Y\mid X,W}(y \mid x, 0)\bigl(1 + \epsilon c_0(y,x)\bigr).
\]

The score takes the form:
\begin{align}
  s(Z) = W c_1(Y,X) + (1-W) c_0(Y,X)
\end{align}
,with $c_w$ satisfying $\mathbb{E}[c_w\mid X,W=w]=0$ for $w=0,1$.

I do not use homoskedasticity assumption to put a constraint on the tangent space. So, I want the estimate to be efficient in a broader class, which includes any form of variance. However, we chose the weights that give the optimal for a homoskedastic DGP.


\section*{Derivation of the Efficient Influence Function}

Since $\tau = \log\mu_1 - \log\mu_0$,
\begin{equation}
  \label{eq:delta}
  \frac{d}{d\epsilon}\tau(P_\epsilon)\bigg|_0
  = \frac{1}{\mu_1}\frac{d}{d\epsilon}\mu_1(P_\epsilon)\bigg|_0
  - \frac{1}{\mu_0}\frac{d}{d\epsilon}\mu_0(P_\epsilon)\bigg|_0 .
\end{equation}

Write $\Psi(Z) := \dfrac{W}{e(X)\,\mu_1} - \dfrac{1-W}{(1-e(X))\,\mu_0}$, with $\mu_0,\mu_1$ evaluated at $\epsilon=0$.  Then

\[
  \frac{d}{d\epsilon}\,\tau(P_\epsilon)\bigg|_0
  = \underbrace{\frac{d}{d\epsilon}\,\mathbb{E}_{P_\epsilon}\bigl[\phi(X,P_0)\,\Psi(Z)\,Y\bigr]\bigg|_0}_{\text{IPW term}}
  + \underbrace{\mathbb{E}_P\!\left[\frac{d\phi(X,P_\epsilon)}{d\epsilon}\bigg|_0\,\Psi(Z)\,Y\right]}_{\text{weight adjustment}} .
\]

In the IPW term the weights do not depend on $\epsilon$, so the derivative acts only on the conditional outcome distribution:

\[\underbrace{\frac{d}{d\epsilon}\,\mathbb{E}_{P_\epsilon}\bigl[\phi(X,P_0)\,\Psi(Z)\,Y\bigr]\bigg|_0}_{\text{IPW term}}
= \mathbb{E}_{P_0}\!\left[\phi(X)\,\Psi(Z)\,Y\cdot s(Z)\right] \]\[
= \mathbb{E}_{P_0}\!\left[\phi(X)\,Y\!\left(\frac{W c_1(Y,X)}{e(X)\,\mu_1} - \frac{(1-W) c_0(Y,X)}{(1-e(X))\,\mu_0}\right)\right].\]

I now, adjusting the EIF to belong to the tangent space:
\[
  \mathrm{EIF}^{(0)}_\tau(Z)
  := \phi(X)\!\left[\frac{W\bigl(Y-m_1(X)\bigr)}{e(X)\,\mu_1}
    - \frac{(1-W)\bigl(Y-m_0(X)\bigr)}{(1-e(X))\,\mu_0}\right].
\]


\subsection*{$\phi(X,P)=\phi(X,\tau,\tau(\cdot))$}

We now restrict the dependence of the weight on $P$ to the parametrization announced in the setup, $\phi(x) = \phi(x,\tau,\tau(\cdot))$, with derivatives $\phi_\tau'$ and $\phi_{\tau(\cdot)}'$.  The chain rule along the submodel $P_\epsilon$ gives
\begin{equation}
  \label{eq:phi_dot}
  \frac{d}{d\epsilon}\phi(X,P_\epsilon)\bigg|_0
  = \phi_\tau'(X)\,\dot\tau + \phi_{\tau(\cdot)}'(X)\,\dot\tau(X),
\end{equation}
where $\dot\tau := \frac{d\tau(P_\epsilon)}{d\epsilon}|_0$ and $\dot\tau(x) := \frac{d\tau(x;P_\epsilon)}{d\epsilon}|_0$.  Substituting \eqref{eq:phi_dot} into the weight-adjustment expectation splits it into two pieces:
\[
  \text{weight adjustment}
  = \underbrace{\dot\tau\;\mathbb{E}\!\left[\phi_\tau'(X)\,\Psi(Z)\,Y\right]}_{T_\tau}
  + \underbrace{\mathbb{E}\!\left[\phi_{\tau(\cdot)}'(X)\,\dot\tau(X)\,\Psi(Z)\,Y\right]}_{T_{\tau(\cdot)}}.
\]

\paragraph{$T_\tau$}
Conditioning on $X$ and using $\mathbb{E}[WY/e(X)\mid X]=m_1(X)$ and $\mathbb{E}[(1-W)Y/(1-e(X))\mid X]=m_0(X)$,
\begin{equation}
  \label{eq:Delta}
  \mathbb{E}\!\left[\Psi(Z)\,Y\mid X\right]
  = \frac{m_1(X)}{\mu_1} - \frac{m_0(X)}{\mu_0}
  =: \Delta(X)
  = \frac{m_1(X)-e^{\tau}m_0(X)}{\mu_1},
\end{equation}
so that
\begin{equation}
  \label{eq:T_tau}
  T_\tau = A\,\dot\tau,\qquad
  A := \mathbb{E}\!\left[\phi_\tau'(X)\,\Delta(X)\right].
\end{equation}

\paragraph{$T_{\tau(\cdot)}$.}
Differentiating $\tau(x) = \log m_1(x) - \log m_0(x)$ along the submodel,
\[
  \dot\tau(x) = \frac{\dot m_1(x)}{m_1(x)} - \frac{\dot m_0(x)}{m_0(x)}.
\]
The submodel's score $s(Z)=W c_1(Y,X) + (1-W) c_0(Y,X)$ with $\mathbb{E}[c_w\mid X,W=w]=0$ gives
\[
  \dot m_w(x) = \mathbb{E}[(Y-m_w(x))\,c_w(Y,x)\mid X=x,W=w].
\]
We rewrite $\mathbb{E}[h(X)\,\dot m_w(X)]$ in two steps.  \emph{Step 1 (inverse-propensity reweighting).}  For any $g(Y,X)$, $\mathbb{E}[g\mid X,W=1] = \mathbb{E}[W g/e(X)\mid X]$ and analogously for $W=0$, so
\[
  \mathbb{E}[h(X)\,\dot m_1(X)] = \mathbb{E}\!\left[h(X)\,\tfrac{W(Y-m_1(X))}{e(X)}\,c_1(Y,X)\right], \quad
  \mathbb{E}[h(X)\,\dot m_0(X)] = \mathbb{E}\!\left[h(X)\,\tfrac{(1-W)(Y-m_0(X))}{1-e(X)}\,c_0(Y,X)\right].
\]
\emph{Step 2 (score identity).}  Since $W^2=W$ and $W(1-W)=0$, multiplying the score by the indicator collapses it to a single component: $W\,s(Z) = W\,c_1(Y,X)$ and $(1-W)\,s(Z) = (1-W)\,c_0(Y,X)$.  Substituting,
\begin{align*}
  \mathbb{E}[h(X)\,\dot m_1(X)]
  &= \mathbb{E}\!\left[h(X)\,\frac{W\,(Y-m_1(X))}{e(X)}\,s(Z)\right],\\
  \mathbb{E}[h(X)\,\dot m_0(X)]
  &= \mathbb{E}\!\left[h(X)\,\frac{(1-W)(Y-m_0(X))}{1-e(X)}\,s(Z)\right].
\end{align*}
Applying these with $h/m_1$ and $h/m_0$ in place of $h$,
\begin{equation}
  \label{eq:Xi_tilde_identity}
  \mathbb{E}\!\left[h(X)\,\dot\tau(X)\right]
  = \mathbb{E}\!\left[h(X)\,\tilde\Xi(Z)\,s(Z)\right],
\end{equation}
where
\[
  \tilde\Xi(Z)
  := \frac{W\,(Y-m_1(X))}{e(X)\,m_1(X)} - \frac{(1-W)(Y-m_0(X))}{(1-e(X))\,m_0(X)}.
\]

Now $T_{\tau(\cdot)} = \mathbb{E}[\phi_{\tau(\cdot)}'(X)\,\dot\tau(X)\,\Psi(Z)\,Y]$, and conditioning on $X$ as in $T_\tau$ replaces $\Psi(Z)Y$ by $\Delta(X)$,
\begin{equation}
  \label{eq:T_taux}
  T_{\tau(\cdot)} = \mathbb{E}\!\left[\phi_{\tau(\cdot)}'(X)\,\Delta(X)\,\dot\tau(X)\right] = \mathbb{E}\!\left[\phi_{\tau(\cdot)}'(X)\,\Delta(X)\,\tilde\Xi(Z)\,s(Z)\right].
\end{equation}

\paragraph{Solving the resulting fixed-point equation.}
Plugging \eqref{eq:T_tau}--\eqref{eq:T_taux} back,
\[
  \dot\tau
  = \mathbb{E}\!\left[\mathrm{EIF}^{(0)}_\tau(Z)\,s(Z)\right]
    + A\,\dot\tau
    + \mathbb{E}\!\left[\phi_{\tau(\cdot)}'(X)\,\Delta(X)\,\tilde\Xi(Z)\,s(Z)\right].
\]
The scalar $\dot\tau$ appears on both sides; isolating it,
\[
  (1 - A)\,\dot\tau
  = \mathbb{E}\!\left[\Bigl(\mathrm{EIF}^{(0)}_\tau(Z) + \phi_{\tau(\cdot)}'(X)\,\Delta(X)\,\tilde\Xi(Z)\Bigr)\,s(Z)\right].
\]
Reading off the influence function gives the closed-form result:
\begin{equation}
  \label{eq:eif_full}
  \boxed{\;
  \mathrm{EIF}_\tau(Z)
  = \frac{1}{C}\Bigl\{\mathrm{EIF}^{(0)}_\tau(Z) + \phi_{\tau(\cdot)}'(X)\,\Delta(X)\,\tilde\Xi(Z)\Bigr\},\;}
\end{equation}
with
\begin{equation}
  \label{eq:C}
  C := 1 - A
     = 1 - \mathbb{E}\!\left[\phi_\tau'(X)\,\Delta(X)\right]
     = 1 - \mathbb{E}\!\left[\phi_\tau'(X)\!\left(\frac{m_1(X)}{\mu_1}-\frac{m_0(X)}{\mu_0}\right)\right].
\end{equation}

\begin{assumption}[Regularity]
$m_0(x)>0$ almost surely, $\mu_0\in(0,\infty)$, and $C\neq 0$ in \eqref{eq:C}.
\end{assumption}

\section*{Minimax optimal weights}

Throughout this section $\tau$ denotes the root of
\begin{equation}
  \label{eq:pseudotrue}
  \mathbb{E}\bigl[\phi_s(X)\{m_1(X)-e^{s}m_0(X)\}\bigr]=0,
  \qquad
  \frac{1}{\phi_s(x)}:=\frac{1}{e(x)}+\frac{e^{s}}{1-e(x)},
\end{equation}
which is the defining relation $\tau=\log(\mu_1/\mu_0)$ evaluated at $\phi=\phi_\tau$.  Set
\begin{equation}
  \label{eq:LambdaQh}
  \Lambda_P(x):=\frac{e^{\tau(x)-\tau}}{e(x)}+\frac{e^{\tau}}{1-e(x)},
  \qquad
  \frac{dQ}{dP}:=\frac{\phi_\tau(X)\,m_0(X)}{\mu_0},
  \qquad
  h(x):=\frac{\phi_\tau(x)}{e(x)}=\frac{1-e(x)}{(1-e(x))+e(x)e^{\tau}} .
\end{equation}

\begin{definition}
$\mathcal{F}$ is the set of measurable $\varphi:\mathbb{R}^p\times\mathbb{R}\to(0,\infty)$ with $\varphi(\cdot,s)\in L^2(m_0\Lambda_P)$.  For $\varphi\in\mathcal{F}$, $\tau_\varphi(P)$ solves $\mathbb{E}[\varphi(X,s)\{m_1(X)-e^{s}m_0(X)\}]=0$, and the \emph{implemented weight} is $\vartheta_{\varphi,P}:=\varphi(\cdot,\tau_\varphi(P))$.
\end{definition}

\begin{definition}
For measurable $\vartheta:\mathbb{R}^p\to(0,\infty)$,
\[
  V(\vartheta;P):=\frac{\sigma^2\,\mathbb{E}\bigl[\vartheta^2m_0\Lambda_P\bigr]}{e^{\tau}\,\mathbb{E}[\vartheta m_0]^2},
  \qquad
  R(\varphi;P):=\frac{V(\vartheta_{\varphi,P};P)}{\inf_\vartheta V(\vartheta;P)}.
\]
\end{definition}

\begin{definition}
$\mathcal{N}_\delta:=\bigl\{P:\ \mathbb{E}_Q[(\tau(X)-\tau)^2]\le\delta^2\bigr\}$.
\end{definition}

\begin{theorem}
\label{thm:minimax}
Under Assumptions 1--4:
\begin{enumerate}[label=(\roman*)]
\item $\inf_\vartheta V(\vartheta;P)=\sigma^2/\bigl(e^{\tau}\mathbb{E}[m_0/\Lambda_P]\bigr)$, attained iff $\vartheta\propto\Lambda_P^{-1}$, and
\[
  R(\varphi;P)
  = \frac{\mathbb{E}\bigl[\vartheta_{\varphi,P}^2m_0\Lambda_P\bigr]\;\mathbb{E}\bigl[m_0/\Lambda_P\bigr]}{\mathbb{E}[\vartheta_{\varphi,P}m_0]^2}
  \;\ge\;1,
\]
with equality iff $\vartheta_{\varphi,P}\propto\Lambda_P^{-1}$.  In particular $R$ is unchanged by any reparametrization $g(\tau)$ with $g'(\tau)\neq0$.
\item If $\tau(\cdot)\equiv s$ then $\Lambda_P^{-1}=\phi_s$, and
\[
  \inf_{\varphi\in\mathcal{F}}\ \sup_{s\in\mathbb{R}}\ \sup_{m_0}\ R(\varphi;P)=1,
\]
attained iff $\varphi(\cdot,s)\propto\phi_s$ for almost every $s$.
\item $R(\phi_\tau;P)=1+\mathbb{V}_Q\bigl[(\tau(X)-\tau)\,h(X)\bigr]+O(\delta^3)$, and
\begin{equation}
  \label{eq:minimaxvalue}
  \inf_{\varphi\in\mathcal{F}}\ \sup_{s\in\mathbb{R}}\ \sup_{m_0}\ \sup_{P\in\mathcal{N}_\delta}\ R(\varphi;P)
  \;=\;1+\delta^2\Bigl(\operatorname*{ess\,sup}_Q h\Bigr)^2+O(\delta^3),
\end{equation}
attained iff $\varphi(\cdot,s)\propto\phi_s$ for almost every $s$.  Under Assumption 2, $\operatorname*{ess\,sup}_Q h\le(1-\eta)/\bigl((1-\eta)+\eta e^{\tau}\bigr)\le1$.
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{(i)} For $\vartheta$ a function of $x$ alone, $\phi_\tau'=\phi_{\tau(\cdot)}'=0$, so $A=0$ and $C=1$ in \eqref{eq:C}, and \eqref{eq:eif_full} reduces to $\mathrm{EIF}_\tau=\mathrm{EIF}^{(0)}_\tau$.  Squaring, the cross term vanishes because $W(1-W)=0$, and conditioning on $(X,W)$,
\[
  \mathbb{E}[\mathrm{EIF}_\tau^2]
  = \mathbb{E}\!\left[\vartheta^2\!\left(\frac{\mathbb{V}[Y\mid X,1]}{e\,\mu_1^2}+\frac{\mathbb{V}[Y\mid X,0]}{(1-e)\,\mu_0^2}\right)\right].
\]
Assumption 3 gives $\mathbb{V}[Y\mid X,w]=\sigma^2m_w$, and $m_1=m_0e^{\tau(X)}$, $\mu_1=e^{\tau}\mu_0$, so the bracket equals
\[
  \frac{\sigma^2m_0}{\mu_0^2}\left(\frac{e^{\tau(X)-2\tau}}{e}+\frac{1}{1-e}\right)
  = \frac{\sigma^2m_0\Lambda_P}{e^{\tau}\mu_0^2},
\]
i.e.\ $\mathbb{E}[\mathrm{EIF}_\tau^2]=V(\vartheta;P)$ with $\mu_0=\mathbb{E}[\vartheta m_0]$.  With $g:=m_0\Lambda_P$,
\[
  \mathbb{E}[\vartheta m_0]^2=\mathbb{E}\!\left[\bigl(\vartheta\sqrt g\bigr)\bigl(m_0/\sqrt g\bigr)\right]^2
  \le \mathbb{E}\bigl[\vartheta^2g\bigr]\,\mathbb{E}\bigl[m_0^2/g\bigr],
  \qquad m_0^2/g=m_0/\Lambda_P,
\]
with equality iff $\vartheta\propto m_0/g=\Lambda_P^{-1}$.  Rearranging gives the bound on $V$, and dividing gives the displayed $R$, in which the factor $e^{\tau}$ cancels.  Any reparametrization multiplies numerator and denominator of $R$ by the same $g'(\tau)^2$, since $\tau$ in \eqref{eq:LambdaQh} is a functional of $P$ alone.

\emph{(ii)} If $\tau(\cdot)\equiv s$ then $m_1=e^{s}m_0$, so $\tau_\varphi(P)=s$ for every $\varphi$, and
$\Lambda_P=1/e+e^{s}/(1-e)=1/\phi_s$ by \eqref{eq:pseudotrue}, which does not involve $m_0$.  By (i), $R\ge1$ with equality iff $\varphi(\cdot,s)\propto\phi_s$; since $\varphi(\cdot,s)$ may be chosen separately for each $s$ and the condition is free of $m_0$, the supremum equals $1$ exactly under that condition.

\emph{(iii)} Write $\langle u,v\rangle_P:=\mathbb{E}[uv\,m_0\Lambda_P]$ and $\psi_P:=\Lambda_P^{-1}$, so that $\mathbb{E}[\vartheta^2m_0\Lambda_P]=\|\vartheta\|_P^2$, $\mathbb{E}[m_0/\Lambda_P]=\|\psi_P\|_P^2$ and $\mathbb{E}[\vartheta m_0]=\langle\vartheta,\psi_P\rangle_P$.  Then (i) reads
\begin{equation}
  \label{eq:Rangle}
  R=\frac{\|\vartheta\|_P^2\,\|\psi_P\|_P^2}{\langle\vartheta,\psi_P\rangle_P^2},
\end{equation}
and since $R$ is invariant under $\vartheta\mapsto c\vartheta$ we may write $\vartheta=\psi_P+u$ with $\langle u,\psi_P\rangle_P=0$, giving $\|\vartheta\|_P^2=\|\psi_P\|_P^2+\|u\|_P^2$, $\langle\vartheta,\psi_P\rangle_P=\|\psi_P\|_P^2$ and
\begin{equation}
  \label{eq:Rexact}
  R=1+\frac{\|u\|_P^2}{\|\psi_P\|_P^2}.
\end{equation}

Let $\Lambda_0:=\Lambda_P$ at $\tau(\cdot)\equiv\tau$ and $\psi_0:=\Lambda_0^{-1}=\phi_\tau$.  Only the first term of $\Lambda_P$ depends on $\tau(\cdot)$, so with $v(x):=\tau(x)-\tau$,
\[
  \Lambda_P-\Lambda_0=\frac{e^{v}-1}{e}=\frac{v}{e}+O(\delta^2),
  \qquad
  \psi_0-\psi_P=\frac{\Lambda_P-\Lambda_0}{\Lambda_0\Lambda_P}
  =\frac{v}{e\,\Lambda_0}\,\psi_0+O(\delta^2)
  =w\,\psi_0+O(\delta^2),
\]
where $w:=v\,h$, using $1/(e\Lambda_0)=\phi_\tau/e=h$.  Since $\psi_0^2m_0\Lambda_0=m_0/\Lambda_0=\phi_\tau m_0=\mu_0\,dQ/dP$, for any $v'$,
\[
  \|v'\psi_0\|_0^2=\mu_0\,\mathbb{E}_Q[v'^2],
  \qquad
  \langle v'\psi_0,\psi_0\rangle_0=\mu_0\,\mathbb{E}_Q[v'],
  \qquad
  \|\psi_0\|_0^2=\mu_0 .
\]
Taking $\vartheta=\psi_0$ in \eqref{eq:Rexact}, $u$ is the component of $\psi_0-\psi_P$ orthogonal to $\psi_P$, and the three displays give
\[
  \frac{\|u\|_P^2}{\|\psi_P\|_P^2}=\mathbb{E}_Q[w^2]-\mathbb{E}_Q[w]^2+O(\delta^3)=\mathbb{V}_Q[w]+O(\delta^3),
\]
which is the first claim of (iii).

For the value, $\mathbb{V}_Q[w]\le\mathbb{E}_Q[v^2h^2]\le\delta^2(\operatorname*{ess\,sup}_Qh)^2$.  Taking $v=\delta(\mathbf{1}_A-Q(A))\{Q(A)(1-Q(A))\}^{-1/2}$ with $Q(A)\to0$ and $A$ concentrating where $h$ approaches its essential supremum gives $\mathbb{E}_Q[w]\to0$ and $\mathbb{E}_Q[w^2]\to\delta^2(\operatorname*{ess\,sup}_Qh)^2$, so the bound is approached.

For optimality, by (ii) any $\varphi$ with $\sup R=1+O(\delta^2)$ has $\vartheta_{\varphi,P}=\psi_0+u_0$ with $\langle u_0,\psi_0\rangle_0=0$ and $\|u_0\|_0=O(\delta)$.  Then $\vartheta_{\varphi,P}-\psi_P=u_0+(\psi_0-\psi_P)+O(\delta^2)$, and $\psi_0-\psi_P$ changes sign when $v$ does, while $\mathcal{N}_\delta$ is invariant under that reflection.  Applying $\|a+b\|^2+\|a-b\|^2=2\|a\|^2+2\|b\|^2$ to $a=u_0$, $b=\psi_0-\psi_P$,
\[
  \max_\pm\bigl\|u_0\pm(\psi_0-\psi_P)\bigr\|_0^2\ \ge\ \|u_0\|_0^2+\|\psi_0-\psi_P\|_0^2,
\]
so by \eqref{eq:Rexact} the supremum over $\mathcal{N}_\delta$ is minimised at $u_0=0$, i.e.\ at $\varphi(\cdot,s)\propto\phi_s$.

Finally $\partial_e h=-e^{\tau}/\bigl((1-e)+e\,e^{\tau}\bigr)^2<0$, so $h$ is decreasing in $e$ and Assumption 2 gives $h\le(1-\eta)/\bigl((1-\eta)+\eta e^{\tau}\bigr)$, which is at most $1$ since $\eta e^{\tau}>0$.
\end{proof}

\begin{proposition}
At $\varphi=\phi_\tau$ and $\tau(\cdot)\equiv\tau$,
\[
  V(\phi_\tau;P)=\frac{\sigma^2}{\mu_1}=\frac{\sigma^2}{\mathbb{E}[\phi_\tau(X)\,m_1(X)]} .
\]
\end{proposition}

\begin{proof}
By (i) and (ii) of Theorem \ref{thm:minimax}, $V=\sigma^2/\bigl(e^{\tau}\mathbb{E}[m_0/\Lambda_0]\bigr)$ with $m_0/\Lambda_0=\phi_\tau m_0$, so $V=\sigma^2/(e^{\tau}\mu_0)=\sigma^2/\mu_1$.  At $\tau(\cdot)\equiv\tau$, $m_1=e^{\tau}m_0$, so $\mu_1=\mathbb{E}[\phi_\tau m_1]$.
\end{proof}
